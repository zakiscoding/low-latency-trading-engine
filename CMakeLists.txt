cmake_minimum_required(VERSION 3.25)
project(lle LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(LLE_NATIVE "Build with -march=native" ON)
option(LLE_LTO "Link-time optimisation" ON)
set(LLE_PGO "" CACHE STRING "PGO mode: '' | gen | use")
set(LLE_PGO_DIR "${CMAKE_BINARY_DIR}/pgo" CACHE PATH "Profile directory")
set(LLE_SANITIZE "" CACHE STRING "address | thread | undefined | ''")
option(LLE_FUZZ "Build libFuzzer targets (clang only)" OFF)
option(LLE_TESTS "Build tests" ON)
option(LLE_URING "Build the io_uring receive path" ON)
set(LLE_OPT_LEVEL "3" CACHE STRING "Optimisation level for Release builds")

add_library(lle_flags INTERFACE)
target_compile_options(lle_flags INTERFACE -Wall -Wextra -Wno-unused-parameter)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(lle_flags INTERFACE -O${LLE_OPT_LEVEL} -fno-plt)
endif()
if(LLE_NATIVE)
  target_compile_options(lle_flags INTERFACE -march=native -mtune=native)
else()
  target_compile_options(lle_flags INTERFACE -march=x86-64-v2)
endif()
if(LLE_LTO AND NOT LLE_SANITIZE)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_msg)
  if(ipo_ok)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
  endif()
endif()
if(LLE_PGO STREQUAL "gen")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(lle_flags INTERFACE -fprofile-instr-generate=${LLE_PGO_DIR}/%p.profraw)
    target_link_options(lle_flags INTERFACE -fprofile-instr-generate)
  else()
    target_compile_options(lle_flags INTERFACE -fprofile-generate -fprofile-dir=${LLE_PGO_DIR} -fprofile-update=single)
    target_link_options(lle_flags INTERFACE -fprofile-generate)
  endif()
elseif(LLE_PGO STREQUAL "use")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(lle_flags INTERFACE -fprofile-instr-use=${LLE_PGO_DIR}/merged.profdata)
  else()
    target_compile_options(lle_flags INTERFACE -fprofile-use -fprofile-dir=${LLE_PGO_DIR} -fprofile-correction
                                               -Wno-missing-profile)
  endif()
endif()
if(LLE_SANITIZE)
  target_compile_options(lle_flags INTERFACE -fsanitize=${LLE_SANITIZE} -fno-omit-frame-pointer -g -O1)
  target_link_options(lle_flags INTERFACE -fsanitize=${LLE_SANITIZE})
endif()

find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)

add_library(lle INTERFACE)
target_include_directories(lle INTERFACE ${CMAKE_SOURCE_DIR}/engine/include)
target_link_libraries(lle INTERFACE lle_flags Threads::Threads)

if(LLE_URING)
  find_library(URING_LIB uring)
  if(URING_LIB)
    target_compile_definitions(lle INTERFACE LLE_HAVE_URING=1)
    target_link_libraries(lle INTERFACE ${URING_LIB})
  endif()
endif()

# ---- binaries (one .cpp each) ----
set(LLE_BINS itch_engine itch_replay itch_synth paper_exchange crypto_engine recorder)
foreach(t ${LLE_BINS})
  if(EXISTS ${CMAKE_SOURCE_DIR}/engine/src/${t}_main.cpp)
    add_executable(${t} engine/src/${t}_main.cpp)
    target_link_libraries(${t} PRIVATE lle OpenSSL::SSL OpenSSL::Crypto)
  endif()
endforeach()
file(GLOB LLE_BENCH_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/engine/bench/*.cpp)
foreach(src ${LLE_BENCH_SRCS})
  get_filename_component(t ${src} NAME_WE)
  add_executable(${t} ${src})
  target_link_libraries(${t} PRIVATE lle)
endforeach()

# ---- tests ----
if(LLE_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  file(GLOB LLE_TEST_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/engine/tests/*.cpp)
  add_executable(lle_tests ${LLE_TEST_SRCS})
  target_link_libraries(lle_tests PRIVATE lle GTest::gtest_main OpenSSL::SSL OpenSSL::Crypto)
  include(GoogleTest)
  gtest_discover_tests(lle_tests DISCOVERY_TIMEOUT 60)
endif()

# ---- fuzzers ----
if(LLE_FUZZ)
  file(GLOB LLE_FUZZ_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/engine/fuzz/*.cpp)
  foreach(src ${LLE_FUZZ_SRCS})
    get_filename_component(f ${src} NAME_WE)
    add_executable(${f} ${src})
    target_include_directories(${f} PRIVATE ${CMAKE_SOURCE_DIR}/engine/include)
    target_link_libraries(${f} PRIVATE OpenSSL::SSL OpenSSL::Crypto)
    target_compile_options(${f} PRIVATE -fsanitize=fuzzer,address,undefined -g -O1)
    target_link_options(${f} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
endif()
